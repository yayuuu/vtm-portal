cmake_minimum_required(VERSION 3.16)
project(xdg-desktop-portal-vtm VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0>=2.70 gio-unix-2.0>=2.70)
option(WITH_SYSTEMD "Install the optional systemd user unit (D-Bus works without it)" ON)
if(WITH_SYSTEMD)
  set(PORTAL_SYSTEMD_SERVICE_LINE "SystemdService=xdg-desktop-portal-vtm.service")
else()
  set(PORTAL_SYSTEMD_SERVICE_LINE "")
endif()
set(SYSTEMD_USER_UNIT_DIR "${CMAKE_INSTALL_PREFIX}/lib/systemd/user" CACHE PATH
    "Directory for systemd user units")
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PORTAL_HELPER_PATH "${CMAKE_INSTALL_FULL_LIBEXECDIR}/vtm-portal-helper")
set(PORTAL_CONFIG_DIR "${CMAKE_INSTALL_FULL_SYSCONFDIR}/xdg-desktop-portal-vtm")
configure_file(src/paths.h.in generated/paths.h @ONLY)
configure_file(data/org.freedesktop.impl.portal.desktop.vtm.service.in
               org.freedesktop.impl.portal.desktop.vtm.service @ONLY)
configure_file(data/xdg-desktop-portal-vtm.service.in xdg-desktop-portal-vtm.service @ONLY)
add_library(portal-common STATIC src/common.cpp src/picker.cpp)
target_include_directories(portal-common PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated"
                           PRIVATE third_party/tomlplusplus)
target_link_libraries(portal-common PUBLIC PkgConfig::GIO)
target_compile_options(portal-common PRIVATE -Wall -Wextra -Wpedantic)
foreach(name vtm-portal-helper vtm-open vtm-run)
  add_executable(${name} src/helper.cpp)
  target_link_libraries(${name} PRIVATE portal-common)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
add_executable(xdg-desktop-portal-vtm src/portal.c)
target_include_directories(xdg-desktop-portal-vtm PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(xdg-desktop-portal-vtm PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(xdg-desktop-portal-vtm PRIVATE PkgConfig::GIO)
install(TARGETS xdg-desktop-portal-vtm vtm-portal-helper RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
install(TARGETS vtm-open vtm-run RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/config.toml DESTINATION ${PORTAL_CONFIG_DIR})
install(FILES data/vtm.portal DESTINATION ${CMAKE_INSTALL_DATADIR}/xdg-desktop-portal/portals)
install(FILES data/vtm-portals.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/xdg-desktop-portal)
install(FILES data/vtm-browser.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.freedesktop.impl.portal.desktop.vtm.service"
        DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/services)
if(WITH_SYSTEMD)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/xdg-desktop-portal-vtm.service"
          DESTINATION ${SYSTEMD_USER_UNIT_DIR})
endif()
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES third_party/tomlplusplus/LICENSE
        DESTINATION ${CMAKE_INSTALL_DOCDIR}/third-party/tomlplusplus)
install(FILES data/vtm-session.sh DESTINATION ${CMAKE_INSTALL_DATADIR}/xdg-desktop-portal-vtm
        RENAME session.sh)
if(BUILD_TESTING)
  find_program(DBUS_RUN_SESSION dbus-run-session)
  if(NOT DBUS_RUN_SESSION)
    message(FATAL_ERROR "Tests need dbus-run-session; use -DBUILD_TESTING=OFF to build without tests")
  endif()
  add_executable(test-helpers tests/helpers.cpp)
  target_link_libraries(test-helpers PRIVATE portal-common)
  add_test(NAME helpers COMMAND test-helpers $<TARGET_FILE:vtm-portal-helper>)
  add_executable(test-protocol tests/protocol.cpp)
  target_link_libraries(test-protocol PRIVATE portal-common)
  add_test(NAME integration COMMAND "${DBUS_RUN_SESSION}" -- $<TARGET_FILE:test-protocol>
           $<TARGET_FILE:xdg-desktop-portal-vtm>)
  set_tests_properties(helpers integration PROPERTIES TIMEOUT 45)
endif()
